### Technical Architecture

```python
class SaaSCopilot:
    """
    RAG (Retrieval Augmented Generation) system that:
    1. Converts user question to structured data query
    2. Fetches relevant data from SaaSIQ database
    3. Generates natural language response with citations
    """
    
    def __init__(self):
        self.llm = OpenAI(model="gpt-4o")  # Or Claude 3.5 Sonnet
        self.query_planner = QueryPlanner()  # Converts NL to SQL/API calls
        self.response_generator = ResponseGenerator()
        self.action_engine = ActionEngine()  # Can TAKE actions, not just answer
    
    async def chat(self, user_message: str, org_id: str) -> CopilotResponse:
        
        # Step 1: Understand intent
        intent = await self.classify_intent(user_message)
        # Intents: "spend_query", "usage_query", "user_lookup", "renewal_info",
        #          "compliance_check", "benchmark", "recommendation", "action_request"
        
        # Step 2: Plan data queries
        queries = self.query_planner.plan(intent, user_message, org_id)
        # Generates: SQL queries, API calls, knowledge graph lookups
        
        # Step 3: Execute queries in parallel
        data = await asyncio.gather(*[self.execute(q) for q in queries])
        
        # Step 4: Generate response with data
        response = await self.response_generator.generate(
            question=user_message,
            data=data,
            intent=intent,
            tone="executive_brief",  # Concise, actionable, with numbers
        )
        
        # Step 5: Suggest follow-up actions
        actions = self.action_engine.suggest_actions(intent, data)
        # "Want me to draft a negotiation email?"
        # "Should I create a reclamation ticket?"
        # "Want to schedule a review meeting?"
        
        return CopilotResponse(
            message=response,
            data_citations=data.sources,
            suggested_actions=actions,
            confidence=data.accuracy_score,
        )
```

## 2. Automated License Reclamation Engine

### What It Is
Automatically identifies unused/underused licenses AND takes action — sends warnings, creates tickets, reclaims licenses — without human intervention.

### Why This Is a Revenue Machine
Every license reclaimed = direct ₹ savings the customer can see. If SaaSIQ saves a customer ₹50L/year, they'll happily pay ₹5L/year for SaaSIQ. **ROI is the easiest sales story in SaaS.**

### How It Works

```python
class LicenseReclamationEngine:
    """
    Automated pipeline: Detect → Warn → Wait → Reclaim → Report
    """
    
    # CONFIG (per customer, per app)
    DEFAULT_POLICY = {
        "warning_after_days_inactive": 30,    # Send warning after 30 days unused
        "grace_period_days": 14,              # Employee has 14 days to respond
        "auto_reclaim_after_days": 44,        # Auto-reclaim after 30+14 days
        "exclude_roles": ["C-Suite", "VP"],   # Never auto-reclaim executives
        "exclude_apps": ["Slack", "Email"],   # Never reclaim core tools
        "min_usage_score_to_keep": 20,        # Score <20 = reclamation candidate
        "notify_manager": True,                # CC the manager
        "notify_employee": True,               # Warn the employee first
        "auto_deprovision": False,             # START with manual, upgrade to auto
    }
    
    async def run_daily_scan(self, org_id: str):
        """
        Daily scan: Find all reclamation candidates
        """
        candidates = []
        
        all_user_apps = get_all_user_app_records(org_id)
        
        for record in all_user_apps:
            # Skip excluded apps and roles
            if record.app in self.policy.exclude_apps:
                continue
            if record.user.role in self.policy.exclude_roles:
                continue
            
            # Check usage score
            if record.usage_score < self.policy.min_usage_score_to_keep:
                candidates.append(ReclamationCandidate(
                    user=record.user,
                    app=record.app,
                    usage_score=record.usage_score,
                    days_inactive=record.days_since_last_active,
                    license_tier=record.license_tier,
                    monthly_cost=record.per_seat_cost,
                    annual_savings=record.per_seat_cost * 12,
                ))
        
        # Process candidates through the pipeline
        for candidate in candidates:
            await self.process_candidate(candidate)
    
    async def process_candidate(self, candidate: ReclamationCandidate):
        """
        State machine: DETECTED → WARNED → GRACE_PERIOD → RECLAIMED
        """
        status = get_reclamation_status(candidate)
        
        if status is None:
            # NEW candidate — send first warning
            await self.send_warning(candidate)
            set_reclamation_status(candidate, "WARNED", warned_at=now())
        
        elif status == "WARNED" and days_since(status.warned_at) >= self.policy.grace_period_days:
            # Grace period expired — check if employee started using it
            if candidate.usage_score > self.policy.min_usage_score_to_keep:
                # Employee started using it! Cancel reclamation.
                set_reclamation_status(candidate, "CANCELLED_ACTIVE")
                return
            
            if self.policy.auto_deprovision:
                # AUTO MODE: Deprovision the license
                await self.deprovision_license(candidate)
                set_reclamation_status(candidate, "RECLAIMED")
                await self.notify_reclamation(candidate)
            else:
                # MANUAL MODE: Create ticket for IT admin
                await self.create_reclamation_ticket(candidate)
                set_reclamation_status(candidate, "PENDING_ADMIN_ACTION")
    
    async def send_warning(self, candidate: ReclamationCandidate):
        """
        Send friendly warning to employee via Slack/Email
        """
        message = f"""
        Hi {candidate.user.first_name},

        We noticed you haven't used **{candidate.app.name}** in 
        {candidate.days_inactive} days.

        Your {candidate.app.name} {candidate.license_tier} license costs 
        ₹{candidate.monthly_cost:,.0f}/month.

        **What happens next:**
        • If you still need it → just open {candidate.app.name} once 
          in the next 14 days and this goes away.
        • If you don't need it → no action needed. The license will be 
          reclaimed and available for someone else.
        • If your usage needs changed → reply to this message and we'll 
          adjust your license tier.

        This helps the company save on tools nobody's using. Thanks! 🙏
        """
        
        await send_slack_dm(candidate.user.slack_id, message)
        await send_email(candidate.user.email, 
                        subject=f"Quick check: Still using {candidate.app.name}?",
                        body=message)
    
    def generate_savings_report(self, org_id: str) -> SavingsReport:
        """
        Monthly report showing exactly how much SaaSIQ saved
        """
        reclaimed = get_reclaimed_licenses(org_id, period="last_30_days")
        
        return SavingsReport(
            period="March 2026",
            licenses_reclaimed=len(reclaimed),
            apps_affected=len(set(r.app for r in reclaimed)),
            monthly_savings=sum(r.monthly_cost for r in reclaimed),
            annual_savings=sum(r.annual_cost for r in reclaimed),
            
            # Top savings by app
            top_savings=[
                {"app": "Salesforce", "licenses_reclaimed": 34, "saved": "₹5.1L/month"},
                {"app": "Figma", "licenses_reclaimed": 27, "saved": "₹1.8L/month"},
                {"app": "Jira", "licenses_reclaimed": 15, "saved": "₹0.9L/month"},
            ],
            
            # ROI calculation
            saasiq_cost="₹3L/year",
            savings_generated="₹42L/year",
            roi_multiple="14x",
        )
```
